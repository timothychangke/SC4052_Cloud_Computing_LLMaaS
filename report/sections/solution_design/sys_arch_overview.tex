\begin{tikzpicture}[
    scale=0.6, 
    transform shape,
    font=\sffamily\small,
    >=Stealth,
    component/.style={draw, rounded corners=2pt, align=center,
        minimum height=10mm, minimum width=28mm, fill=blue!5,
        inner sep=3pt},
    llm/.style={component, fill=orange!15, draw=orange!60!black},
    store/.style={draw, cylinder, shape border rotate=90,
        aspect=0.25, align=center, fill=green!10,
        draw=green!50!black, minimum height=12mm, minimum width=22mm},
    gate/.style={draw, diamond, aspect=2, align=center,
        fill=red!10, draw=red!60!black, inner sep=1pt,
        minimum width=20mm},
    flow/.style={->, thick, draw=black!70},
    async/.style={->, thick, dashed, draw=purple!70},
    lbl/.style={font=\sffamily\scriptsize\itshape, text=black!60,
        inner sep=1pt, fill=white}
]

% ===== Row 0: Frontend =====
\node[component, fill=blue!15] (fe) at (0, 7)
    {React Chat UI\\\scriptsize (chat + image + analytics)};

% ===== Row 1: Orchestrator =====
\node[component, fill=blue!20, minimum width=180mm] (orch) at (0, 5)
    {FastAPI Orchestrator\\\scriptsize \texttt{/chat} \quad \texttt{/image} \quad \texttt{/track/click} \quad \texttt{/analytics} \quad \texttt{/portal}};

% ===== Row 2: Sync pipeline =====
\node[llm]       (ctx)   at (-8.5, 2.0) {Context Extraction\\\scriptsize gpt-oss-20b \\\scriptsize \{intent, topics,\\\scriptsize sentiment, signals\}};
\node[gate]      (gate)  at (-4.5, 2.0) {Receptivity\\Gate};
\node[llm]       (emb)   at ( 0.0, 2.0) {Query Embedding\\\scriptsize text-embedding\\\scriptsize -3-small};
\node[component] (retr)  at ( 4.0, 2.0) {Vector Retrieval\\\scriptsize pgvector\\\scriptsize top-20};
\node[component] (rank)  at ( 8.0, 2.0) {Re-ranker\\\scriptsize sim+bid+signal\\\scriptsize freq-cap, budget};
\node[llm]       (synth) at ( 12.5, 2.0) {Response Synthesis\\\scriptsize gpt-oss-120b \\\scriptsize ad injection\\\scriptsize \texttt{[Sponsored]}};

% ===== Image branch (distinct lower-right track) =====
\node[llm, fill=orange!8] (rewrite) at ( 7.0, -1.3)
    {Placement Prompt\\Rewriter (gpt-oss-120b )};
\node[llm, fill=orange!8] (img)     at ( 11.2, -1.3)
    {\texttt{gpt-image-1}\\\scriptsize generate / edit};

% ===== Data stores =====
\node[store] (pg)    at (-8.5, -5.0)
    {PostgreSQL\\+ pgvector\\\scriptsize ads, embeddings,\\\scriptsize events (JSONB)};

\node[store] (redis) at ( 2.0, -3.0)
    {Redis\\\scriptsize Session State\\\scriptsize 10-turn window\\\scriptsize ads\_shown};

% ===== Async feedback row (well below PG to avoid horizontal crossings) =====
\node[llm, fill=purple!8, draw=purple!60] (judge) at ( -2.0, -7.0)
    {Engagement / LLM-Judge\\\scriptsize follow-up classifier\\\scriptsize naturalness scorer};
\node[component, fill=purple!8, draw=purple!60] (opt) at ( 5.5, -7.0)
    {Context Optimiser\\\scriptsize topic decay,\\\scriptsize ad-fatigue update};

% ===== Sync flow arrows =====
\draw[flow] (fe) -- node[lbl, right]{user msg / image} (orch);

% Orchestrator -> Context Extraction
\draw[flow] (orch.south -| ctx) -- (ctx.north);

\draw[flow] (ctx) -- (gate);
\draw[flow] (gate) -- node[lbl, above, yshift=1mm]{receptive} (emb);
\draw[flow] (emb) -- (retr);
\draw[flow] (retr) -- (rank);
\draw[flow] (rank) -- (synth);

% Response Synthesis -> Orchestrator: up-then-left to orch top-right.
\draw[flow] (synth.north) |- (orch.east);
\node[lbl, above] at ($(synth.north |- orch.east)!0.5!(orch.east)$) {response};

% Gate "none => skip" bypass: drop from gate bottom, run right well below
% the pipeline and above Redis, up into Placement Rewriter's south side.
% Path stays below the pipeline but above Redis (y ~ 0.0), avoiding Redis.
\draw[flow] (gate.south) -- ++(0,-1.3) coordinate (skipA) -- (skipA -| rewrite.north) -- (rewrite.north);
\node[lbl, above] at ($(skipA)!0.5!(skipA -| rewrite.north)+(-1.5,0.3)$) {\texttt{none} $\Rightarrow$ skip ads};
% ===== Image branch =====
% Re-ranker -> Rewriter: down-then-left. Label on vertical segment.
\draw[flow] (rank.south) -- ++(0,-1.5) -| (rewrite.north);
\draw[flow] (rewrite) -- (img);

% gpt-image-1 -> Response Synthesis: right-then-up into synth south.
\draw[flow] (img.north) -| (synth.south);

% ===== Stores =====
% Context Extraction <-> PostgreSQL: clean vertical double-arrow.
\draw[flow, <->] (ctx.south) -- (pg.north);
\node[lbl, right] at ($(ctx.south)!0.5!(pg.north)$) {read / write};

% PostgreSQL -> Redis: horizontal from PG-top-right area, but we need to
% avoid going through the judge. Instead, route from PG's TOP-RIGHT
% going up first, then right, then up into Redis west.
\draw[flow] (pg.north east) -- ++(1.0,0) coordinate (psR) -- (psR |- redis.west) -- (redis.west);
\node[lbl, above] at ($(psR |- redis.west)!0.5!(redis.west)$) {ads\_shown};

% Redis <-> Orchestrator: vertical double-arrow, label floated to the side.
\draw[flow, <->] (redis.north) -- (redis.north |- orch.south);
\node[lbl, right] at ($(redis.north)!0.5!(redis.north |- orch.south)+(0.15,1.2)$) {session R/W};

% Re-ranker -> Redis: freq-cap lookup.
\draw[flow] ([xshift=-8mm]rank.south) -- ++(0,-1.0) -| ([xshift=6mm]redis.north);
\node[lbl] at ([xshift=-6mm, yshift=-4mm]rank.south) {freq-cap};

% ===== Async loop =====
% PostgreSQL -> Judge: down from PG, right to judge west. Clean path.
\draw[async] (pg.south) -- ++(0,-0.9) |- (judge.west);
\node[lbl, above] at ($(pg.south)+(0,-0.9)+(1.5,0)$) {events};

\draw[async] (judge) -- (opt);

% Context Optimiser -> Redis: up from opt, left into redis south.
\draw[async] (opt.north) -- ++(0,0.8) coordinate (optA) -- (optA -| redis.south) -- (redis.south);
\node[lbl, right] at ($(optA)!0.3!(optA -| redis.south)+(0.1,0.15)$) {updated context};

% ===== Legend =====
\matrix [draw, rounded corners=2pt, fill=white,
         row sep=2pt, column sep=4pt,
         nodes={font=\scriptsize\sffamily, anchor=west},
         ampersand replacement=\&] at (-8.5, -9.5) {
  \node[component, minimum width=5mm, minimum height=3mm]{}; \& \node{Service}; \&
  \node[llm, minimum width=5mm, minimum height=3mm]{};       \& \node{LLM call}; \\
  \node[store, minimum width=5mm, minimum height=3mm]{};     \& \node{Data store}; \&
  \node[gate, minimum width=5mm, minimum height=3mm]{};      \& \node{Safety gate}; \\
  \draw[flow] (0,0) -- (5mm,0);  \& \node{Sync path}; \&
  \draw[async] (0,0) -- (5mm,0); \& \node{Async loop}; \\
};

\end{tikzpicture}
